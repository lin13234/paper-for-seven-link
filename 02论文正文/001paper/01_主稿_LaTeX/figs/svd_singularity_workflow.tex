% 奇异值分解及判据。
\node (svd) [process] {
    \textbf{Singular Value Decomposition (SVD)}\\[1mm]
    Compute singular values $\sigma_i$ and condition number $\kappa(J_p)$
};

\node (dec1) [decision, below=of svd] {
    Is $\sigma_{\min}(J_p)\leq\varepsilon$?
};

\node (trueSing) [true outcome, right=30mm of dec1] {
    \centering\textbf{True Singular Candidate}\par
    \vspace{1.5mm}
    \raggedright
    $\bullet$ Rank deficiency detected\\
    $\bullet$ Instantaneous DOF increases\\
    $\bullet$ Confirm by branch continuation
};

\node (dec2) [decision, below=16mm of dec1] {
    Is $\sigma_{\min}(J_p)$ a local minimum\\
    and $\sigma_{\min}(J_p)>\varepsilon$?
};

\node (pseudoSing) [pseudo outcome, right=30mm of dec2] {
    \centering\textbf{Ill-conditioned / Pseudo-singular}\par
    \vspace{1.5mm}
    \raggedright
    $\bullet$ Local sensitivity peak\\
    $\bullet$ No branch intersection\\
    $\bullet$ Large condition number $\kappa(J_p)$
};

\node (regular) [regular outcome, below=16mm of dec2] {
    \centering\textbf{Regular Configuration}\par
    \vspace{1.5mm}
    \raggedright
    $\bullet$ $\sigma_{\min}$ remains safely above the threshold\\
    $\bullet$ Well-conditioned passive Jacobian\\
    \centering\textit{Stable motion on the selected branch}
};

% 流程箭头与分支标签。
\draw[flow arrow] (svd) -- (dec1);
\draw[flow arrow] (dec1.east) -- node[branch label, above] {Yes} (trueSing.west);
\draw[flow arrow] (dec1.south) -- node[branch label, right] {No} (dec2.north);
\draw[flow arrow] (dec2.east) -- node[branch label, above] {Yes} (pseudoSing.west);
\draw[flow arrow] (dec2.south) -- node[branch label, right] {No} (regular.north);
